\subsection{Vector-Space Optimization Framework}

\begin{frame}{\insertsubsectionhead}{}

  The contraction condition from Theorem 2.4 \cite{Tsukamoto:2021aa} is represented as follows:
  \begin{equation} \label{eq:cstr:ctrc}
  \ddtt\mm{M}
  +
  2\mysym
  \left(
      \mm{M}
      \mathbb{A}
  \right)
  - 
  2
  \mm{M}\mm{B}\mm{R}^{-1}\mm{B}^\top\mm{M}
  \preceq
  -2\alpha \mm{M}
  .
  \end{equation}

  \hfill

  We note that eventually the metric is used as a feedback gain matrix as 
  \begin{equation} \label{eq:control:law}
    \mv{u}
    =
    \mv{u}_d
    -
    \mm{R}^{-1} \mm{B}^\top \mm{M}
    % ( 
    %    \mv{x} - \mv{x}_d
    % )
    \mv{e}
    .
  \end{equation}
  Therefore, the metric is only used in \cred{the subspace spanned by the error} $\mv{e} = \mv{x} - \mv{x}_d$.

  \hfill

  Thus, we can project the contraction condition onto $\mv{e}$, which results in a \cred{point-wise convex condition} with low computational burden.

  \begin{equation} \label{eq:cstr:ctrc:proj}
    \mv{y}^\top 
    \!
    \left(
    \!
      -\mm{B} \mm{R}^{-1} \mm{B}^\top
    \!
    \right)
    \mv{y}
    +
    \mv{e}^\top
    \!
    \left(
      \tfrac{1}{T_s}+2\alpha
      +
      2\mathbb{A}^\top
    \!
    \right)
    \mv{y}
    -
    \tfrac{\mv{e}^\top \mm{M}_{\rm{pre}} \mv{e}}{T_s}
    \le 0
    ,
  \end{equation}
  where $\mv{y}:=\mm{M}\mv{e}$ denotes a projected contraction metric onto the $\mv{e}$, serving as a new optimization variable.

\end{frame}

% ————————————————————————————————————————————————————————————————
\begin{frame}{\insertsubsectionhead}{Details of the Proposed Method}

  \textbf{Objective Function (Condition Number)}:
  According to Theorem 2.4 in \cite{Tsukamoto:2021aa}, the minimization of the condition number $\chi$ of $\mm{M}$ can be achieved by minimizing the following objective function:

  \begin{equation}
    J = \norm{\mv{y}}\norm{\mv{e}} - \mv{y}^\top \mv{e}
    .
  \end{equation}

    
  \textbf{Contraction Condition}:
  Rewrite the projected contraction condition as

  \begin{equation} \label{eq:cstr:ctrc:proj}
    \mv{y}^\top 
    \!
    \left(
    \!
      -\mm{B} \mm{R}^{-1} \mm{B}^\top
    \!
    \right)
    \mv{y}
    +
    \mv{e}^\top
    \!
    \left(
      \tfrac{1}{T_s}+2\alpha
      +
      2\mathbb{A}^\top
    \!
    \right)
    \mv{y}
    -
    \tfrac{\mv{e}^\top \mm{M}_{\rm{pre}} \mv{e}}{T_s}
    \le 0
    .
  \end{equation}

  \textbf{Energy Condition}: Minimizing the condition number may result in \cred{a very small control input}. To avoid this issue, we add the energy condition as follows:

  \begin{equation} \label{eq:cstr:energy}
  \left(
    \mv{e}^\top \mv{y} 
    -
    \mu \mv{e}^\top \mv{e}
  \right)^2
  \le
  (\epsilon + s) \norm{\mv{e}}^2
  ,
  \end{equation}
  where $\mu\in\R_{>0}$ denotes the \cblue{desired energy level}, and $\epsilon\in\R_{>0}$ and $s\in\R_{\ge0}$ denote the allowable small energy variation and a slack variable.

  \textbf{Additional Constraints}:
  Any constraints can be added to the optimization problem.
 
\end{frame}

% ————————————————————————————————————————————————————————————————
\begin{frame}{\insertsubsectionhead}{}

  \textbf{Resulting Optimization Problem}:

  \begin{equation} \label{eq:opt:main}
    \begin{matrix}
      \min_{
        \mv{y},s
        } 
      \norm{\mv{y}}\norm{\mv{e}} - \mv{y}^\top \mv{e}
      +
      \lambda s
      \\ 
      \vspace{-.2cm}
      \\
      \text{s.t. }
      \begin{cases}
        \mv{y}^\top 
      \!
      \left(
      \!
        -\mm{B} \mm{R}^{-1} \mm{B}^\top
      \!
      \right)
      \mv{y}
      +
      \mv{e}^\top
      \!
      \left(
        \tfrac{1}{T_s}+2\alpha
        +
        2\mathbb{A}^\top
      \!
      \right)
      \mv{y}
      -
      \tfrac{\mv{e}^\top \mm{M}_{\rm{pre}} \mv{e}}{T_s}
      \le 0
      ,
      \\
      \left(
        \mv{e}^\top \mv{y} 
        -
        \mu \mv{e}^\top \mv{e}
      \right)^2
      \le
      (\epsilon + s) \norm{\mv{e}}^2
      ,
      \end{cases}
    \end{matrix}
  \end{equation}

$\lambda$ to achieve sufficient control effort and $\mu$ to determine the desired input energy
