\subsection{Vector-Space Optimization Framework}

\begin{frame}{\insertsubsectionhead}{}

  The contraction condition from Theorem 2.4 \paperinfo{H.~Tsukamoto, S.-J. Chung, and J.-J.~E. Slotine, ``Contraction theory for nonlinear stability analysis and learning-based control: A tutorial overview,'' \emph{Annual Reviews in Control}, 2021} is represented as follows:
  \begin{equation} \label{eq:cstr:ctrc}
  \underbrace{\ddtt\mm{M}}_{\approx(\mm{M}-\mm{M}_{\rm{pre}})/T_s}
  +
  2\mysym
  \left(
      \mm{M}
      \mathbb{A}
  \right)
  - 
  2
  \mm{M}\mm{B}\mm{R}^{-1}\mm{B}^\top\mm{M}
  \preceq
  -2\alpha \mm{M}
  .
  \end{equation}

  \hfill

  \onslide<2->
  We note that eventually the metric $\mm{M}$ is used as a feedback gain matrix as follows:
  \begin{equation} \label{eq:control:law}
    \mv{u}
    =
    \mv{u}_d
    -
    \mm{R}^{-1} \mm{B}^\top \mm{M}
    % ( 
    %    \mv{x} - \mv{x}_d
    % )
    \mv{e}
    .
  \end{equation}
  Therefore, the metric $\mm{M}$ enters the control input only through its action on the error $\mv{e}$.

  \hfill

  \onslide<3->
  Thus, we can project the contraction condition onto $\mv{e}$, which results in the following concave condition:

  \begin{equation} \label{eq:cstr:ctrc:proj}
    \mv{e}^\top
    \left(
    \text{Eq. (15)}
    \right)
    \mv{e}
    \quad
    \rightarrow
    \quad
    \mv{y}^\top 
    \!
    \left(
    \!
      -\mm{B} \mm{R}^{-1} \mm{B}^\top
    \!
    \right)
    \mv{y}
    +
    \mv{e}^\top
    \!
    \left(
      \tfrac{1}{T_s}+2\alpha
      +
      2\mathbb{A}^\top
    \!
    \right)
    \mv{y}
    -
    \tfrac{\mv{e}^\top \mm{M}_{\rm{pre}} \mv{e}}{T_s}
    \le 0
    ,
  \end{equation}
  where $\mv{y}:=\mm{M}\mv{e}$ denotes a \cblue{new decision variable which is a metric-weighted error vector}.

\end{frame}

% ————————————————————————————————————————————————————————————————
\begin{frame}{\insertsubsectionhead}{Details of the Proposed Method}

  \textbf{Obj Func.}:
  Minimization of the condition number $\chi$ of $\mm{M}$

  \begin{equation}
    J = \norm{\mv{y}}\norm{\mv{e}} - \mv{y}^\top \mv{e}
    \quad
    (\text{promotes $\mm{M}$ to be isotropic — minimizing $\chi$})
  \end{equation}
    
  \textbf{Cstr. 1 (Projected) Contraction Constraint}:

  \begin{equation} \label{eq:cstr:ctrc:proj}
    \mv{y}^\top 
    \left(
      -\mm{B} \mm{R}^{-1} \mm{B}^\top
    \right)
    \mv{y}
    +
    \mv{e}^\top
    \left(
      \tfrac{1}{T_s}+2\alpha
      +
      2\mathbb{A}^\top
    \right)
    \mv{y}
    -
    \tfrac{\mv{e}^\top \mm{M}_{\rm{pre}} \mv{e}}{T_s}
    \le 0
    .
  \end{equation}

  \textbf{Cstr. 2 Energy-Based Constraint}: 

  \begin{equation} \label{eq:cstr:energy}
  \left(
    \mv{e}^\top \mv{y} 
    -
    \mu \mv{e}^\top \mv{e}
  \right)^2
  \le
  (\epsilon + s) \norm{\mv{e}}^2
  ,
  \quad
  (\text{
    prevents $\mm{M}$ from being too small by obj. func.
  })
  \end{equation}
  where 
  \begin{itemize}
    \item $\mu\in\R_{>0}$: desired energy level
    \item $\epsilon\in\R_{>0}$: allowable small energy variation
    \item $s\in\R_{\ge0}$: slack variable (for feasibility)
  \end{itemize}

  \textbf{Cstr. 3 Input Constraints}:
  Any constraints can be added to the optimization problem.

  \begin{equation}
    c(\mv{y}) \le 0
  \end{equation}

\end{frame}

% ————————————————————————————————————————————————————————————————
\begin{frame}{\insertsubsectionhead}{}

  \textbf{Resulting Optimization Problem}:

  \begin{equation} \label{eq:opt:main}
    \begin{matrix}
      \min_{
        \mv{y},s
        } 
      \norm{\mv{y}}\norm{\mv{e}} - \mv{y}^\top \mv{e}
      +
      \lambda s
      \\ 
      \vspace{-.2cm}
      \\
      \text{s.t. }
      \begin{cases}
      \ctxt{blue}{
        \mv{y}^\top 
        \!
        \left(
        \!
          -\mm{B} \mm{R}^{-1} \mm{B}^\top
        \!
        \right)
        \mv{y}
        +
        \mv{e}^\top
        \!
        \left(
          \tfrac{1}{T_s}+2\alpha
          +
          2\mathbb{A}^\top
        \!
        \right)
        \mv{y}
        -
        \tfrac{\mv{e}^\top \mm{M}_{\rm{pre}} \mv{e}}{T_s}
        \le 0
      }
      ,
      &
      \text{— \textit{(Projected) contraction constraint}}
      \\
      \ctxt{teal}{
        \left(
          \mv{e}^\top \mv{y} 
          -
          \mu \mv{e}^\top \mv{e}
        \right)^2
        \le
        (\epsilon + s) \norm{\mv{e}}^2
      }
      ,
      &
      \text{— \textit{Energy-based constraint}}
      \\
      \ctxt{red}{
        c(\mv{y}) \le 0
      }
      ,
      &
      \text{— \textit{Input constraint}}
    \end{cases}
    \end{matrix}
  \end{equation}

  \textbf{Without Input Saturation}: 
  (a) is the optimal solution.
  \begin{itemize}
    \item Minimum condition number $\chi$.
    \item Input energy constraint satisfied.
  \end{itemize}

  \vspace{0.5cm}

  \textbf{With Input Saturation}: 
  (b) is the optimal solution.
  \begin{itemize}
    \item Input saturation constraint satisfied.
    \item Minimum condition number under the input constraint.
    \item Slack variable $s$ is non-zero, relaxing energy-based constraint.
